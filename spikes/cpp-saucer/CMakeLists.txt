cmake_minimum_required(VERSION 3.25)
project(cpp_saucer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
# saucer master as spiked on 2026-11-09 (v8.0.0 + 32 commits).
FetchContent_Declare(saucer GIT_REPOSITORY https://github.com/saucer/saucer GIT_TAG c7dca16ca84b6ebde3ec74e8944238740eb38694)
FetchContent_MakeAvailable(saucer)
CPMFindPackage(NAME saucer-desktop VERSION 4.1.0 GIT_REPOSITORY "https://github.com/saucer/desktop")
CPMAddPackage(NAME picosha2 GIT_REPOSITORY "https://github.com/okdshin/PicoSHA2" GIT_TAG master DOWNLOAD_ONLY YES)
add_executable(cpp_saucer main.cpp)
target_include_directories(cpp_saucer PRIVATE ${picosha2_SOURCE_DIR})
target_compile_definitions(cpp_saucer PRIVATE INDEX_HTML="${CMAKE_CURRENT_SOURCE_DIR}/../common/index.html")
target_link_libraries(cpp_saucer PRIVATE saucer saucer::desktop)
